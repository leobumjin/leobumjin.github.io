<script>
  document.querySelectorAll('.project-card-stars[data-repo]').forEach(function (link) {
    var count = link.querySelector('[data-star-count]');
    fetch('https://api.github.com/repos/' + link.dataset.repo)
      .then(function (response) {
        if (!response.ok) throw new Error('GitHub unavailable');
        return response.json();
      })
      .then(function (repo) {
        if (typeof repo.stargazers_count !== 'number') throw new Error('Missing stars');
        count.textContent = repo.stargazers_count.toLocaleString() + (repo.stargazers_count === 1 ? ' star' : ' stars');
      })
      .catch(function () { count.textContent = 'View stars'; });
  });
</script>
